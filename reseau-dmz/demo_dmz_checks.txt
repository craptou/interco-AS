# Vérifications DMZ (commandes)

# Démarrer
cd reseau-dmz
docker compose up -d

# Vérifier les conteneurs
docker ps --filter "name=dmz"

# Vérifier DNS
docker exec dmz-dns cat /etc/dnsmasq.d/hosts
docker run --rm --network=core_net_dmz instrumentisto/dig +short dmz-web.dmz.local @120.0.24.11

# Tester HTTP depuis DMZ
docker run --rm --network=core_net_dmz curlimages/curl:latest curl -s http://120.0.24.10 | head -n 5

# Tester HTTP depuis CORE (doit fonctionner)
docker run --rm --network=core_net_core curlimages/curl:latest curl -s --max-time 5 http://120.0.24.10 | head -n 5

# Tester HTTP depuis ENT (doit être bloqué)
docker run --rm --network=core_net_ent_main curlimages/curl:latest curl -s --max-time 5 http://120.0.24.10 || echo "BLOQUE (attendu)"

# Règles du pare-feu
docker exec dmz-fw nft list ruleset

# Vérifier forwarding
docker exec dmz-fw sysctl net.ipv4.ip_forward

# Ping depuis DMZ
docker run --rm --network=core_net_dmz busybox ping -c 2 120.0.24.10

# Nettoyage
cd reseau-dmz
docker compose down
